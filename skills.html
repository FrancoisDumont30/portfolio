Human Traffic of Stadium

Find runs of at least three consecutive days with 100 visitors or more. Hard level. I solved it in SQL with a CTE and window functions.

SQL (CTEs, LAG, LEAD, UNION ALL)

WITH cte AS (
    SELECT 
    LAG(id) OVER() AS a,
    id,
    LEAD(id) OVER() AS c,
    visit_date,
    people
    FROM Stadium
    WHERE people >= 100
),
    filter AS (
    SELECT 
    *
    FROM cte
    WHERE a = id-1 AND c = id+1)
, aggreg AS (
    SELECT a AS res FROM filter
    UNION ALL
    SELECT id AS res FROM filter
    UNION ALL
    SELECT c AS res FROM filter
)

SELECT * FROM Stadium WHERE id IN (SELECT DISTINCT res FROM aggreg)

Trips and Users

Hard level. Daily cancellation rate of trips between unbanned clients and drivers over three days. Solved in SQL with a CTE.

SQL (CTE, CASE, subqueries, AVG, GROUP BY)

WITH cte AS (
    SELECT *, 
    CASE
        WHEN status = 'completed' THEN 0
        ELSE 1
        END AS Cancellation
    FROM Trips
    WHERE client_id IN (SELECT users_id FROM Users WHERE banned='No')
    AND driver_id IN (SELECT users_id FROM Users WHERE banned='No')
    AND request_at > '2013-09-30' AND request_at < '2013-10-04')
SELECT request_at AS Day, ROUND(AVG(Cancellation), 2) AS `Cancellation Rate`
FROM cte
GROUP BY Day

Consecutive Numbers

Find numbers that appear at least three times in a row, using the previous and next rows.

SQL (CTE, LAG, LEAD)

WITH
    T AS (
        SELECT
            *,
            LAG(num) OVER () AS a,
            LEAD(num) OVER () AS b
        FROM Logs
    )
SELECT DISTINCT num AS ConsecutiveNums
FROM T
WHERE a = num AND b = num;

Rank Scores

Rank scores from highest to lowest, with no gaps between ranks.

SQL (CTE, ROW_NUMBER, window function)

WITH cte AS (SELECT 
    score,
    ROW_NUMBER() OVER (ORDER BY score DESC) AS `ranking`
FROM Scores
GROUP BY score)
SELECT s.score, c.ranking as `rank`
FROM Scores s
JOIN cte c
ON s.score = c.score
ORDER BY score DESC

Students and Examinations

Count how many exams each student attended for every subject, including zero. The key is to build all student and subject pairs first, then join the exams.

SQL (CROSS JOIN, LEFT JOIN, GROUP BY)

SELECT 
    Students.student_id, 
    Students.student_name, 
    Subjects.subject_name, 
    COUNT(Examinations.subject_name) AS attended_exams

FROM Students

CROSS JOIN Subjects

LEFT JOIN Examinations 
    ON Students.student_id = Examinations.student_id 
    AND Subjects.subject_name = Examinations.subject_name

GROUP BY Students.student_id, Subjects.subject_name
ORDER BY Students.student_id, Subjects.subject_name
;

Pandas (cross merge, groupby, fillna)

import pandas as pd

def students_and_examinations(students: pd.DataFrame, subjects: pd.DataFrame, examinations: pd.DataFrame) -> pd.DataFrame:


    df = students.merge(subjects, how='cross')

    exam_counts = examinations.groupby(['student_id', 'subject_name']).size().reset_index(name='attended_exams')

    df = df.merge(exam_counts, on=['student_id', 'subject_name'], how='left')
    df = df[['student_id','student_name','subject_name', 'attended_exams']]
    df['attended_exams'] = df['attended_exams'].fillna(0)

    return df.sort_values(['student_id', 'subject_name'])

Confirmation Rate

Compute each user's share of confirmed messages, with 0 for users who never confirmed anyone.

SQL (LEFT JOIN, AVG, GROUP BY)

SELECT s.user_id, ROUND(AVG(IF(c.action = 'confirmed', 1,0)),2) AS confirmation_rate
FROM Signups s
LEFT JOIN Confirmations c
ON s.user_id = c.user_id
GROUP BY s.user_id

Pandas (replace, groupby mean, outer merge)

import pandas as pd

def confirmation_rate(signups: pd.DataFrame, confirmations: pd.DataFrame) -> pd.DataFrame:
    confirmations = confirmations.replace('confirmed', 1)
    confirmations = confirmations.replace('timeout', 0)
    confirmations = confirmations.groupby('user_id')['action'].mean().round(2).reset_index(name='confirmation_rate')
    
    confirmations = confirmations.merge(signups, on='user_id', how='outer')
    confirmations = confirmations.fillna(0)


    return confirmations[['user_id','confirmation_rate']]
